### Visualization of annotated genome assembly via GenoVi

#### Installation and patching the installation

General instructions on [https://github.com/robotoD/GenoVi](https://github.com/robotoD/GenoVi)

##### Condensed setup (incl. patch)
```bash
# 1) Install in a dedicated conda environment
conda create -n genovi python=3.11 pip circos perl -y -c conda-forge -c bioconda
conda activate genovi
pip install genovi

# If circos is not available via conda on your system, use Debian fallback:
# sudo apt install -y circos perl libgd-perl libsvg-perl libset-intspan-perl \
#   libstatistics-descriptive-perl libtext-format-perl libparams-validate-perl liblist-moreutils-perl

# 2) Apply all required GenoVi runtime fixes at once
python3 - <<'PY'
import importlib.util, pathlib, re

def mod_path(name):
    spec = importlib.util.find_spec(name)
    if spec is None or not spec.origin:
        raise SystemExit(f"Could not locate {name} in this Python environment")
    return pathlib.Path(spec.origin)

def backup_and_read(p):
    b = p.with_suffix(p.suffix + ".bak")
    if not b.exists():
        b.write_text(p.read_text())
    return p.read_text()

# createConf.py: robust etc/fonts staging, remove old duplicate-output override block
p = mod_path("scripts.createConf")
txt = backup_and_read(p)
txt = re.sub(r'\n\s*# Ensure Circos can resolve includes like etc/.*?\n', '\n', txt, flags=re.S)
txt = re.sub(r'\n\s*# Force GenoVi/Circos to generate an SVG.*?pass\n', '\n', txt, flags=re.S)
anchor = 'file = open("circos.conf", "w")'
if anchor in txt and 'CIRCOS_SRC_CANDIDATES' not in txt:
    txt = txt.replace(anchor, anchor + '''
    import os, shutil

    # CIRCOS_SRC_CANDIDATES: probe conda + circos binary prefix + system paths
    conda_prefix = os.environ.get("CONDA_PREFIX", "")
    circos_bin = shutil.which("circos")
    circos_prefix = os.path.abspath(os.path.join(os.path.dirname(circos_bin), "..")) if circos_bin else ""

    candidates = []
    if conda_prefix:
        candidates.extend([
            os.path.join(conda_prefix, "etc", "circos"),
            os.path.join(conda_prefix, "share", "circos"),
        ])
    if circos_prefix:
        candidates.extend([
            os.path.join(circos_prefix, "etc"),
            os.path.join(circos_prefix, "share", "circos"),
        ])
    candidates.extend(["/etc/circos", "/usr/share/circos"])

    src = ""
    for c in candidates:
        if os.path.isfile(os.path.join(c, "fonts", "modern", "cmunssdc.ttf")):
            src = c
            break

    shutil.rmtree("etc", ignore_errors=True)
    shutil.rmtree("fonts", ignore_errors=True)
    if src:
        shutil.copytree(src, "etc", dirs_exist_ok=True)
        fonts_src = os.path.join(src, "fonts")
        if os.path.isdir(fonts_src):
            shutil.copytree(fonts_src, "fonts", dirs_exist_ok=True)
        print(f"Using Circos resources from: {src}")
    else:
        os.makedirs("etc", exist_ok=True)
        os.makedirs("fonts", exist_ok=True)
''', 1)
p.write_text(txt)

# GenoVi.py: fix bad shell redirection and keep debug log
p = mod_path("scripts.GenoVi")
txt = backup_and_read(p)
txt = re.sub(
    r'os\.system\("circos -conf circos\.conf.*?"\)',
    'os.system("circos -conf circos.conf -outputdir . -outputfile circos.svg > circos.log 2>&1")',
    txt
)
txt = txt.replace(
    'os.system("circos -debug_group _all >/dev/null 2>&1")',
    'os.system("circos -debug_group _all > circos.debug.log 2>&1")'
)
p.write_text(txt)

# create_tables.py: matplotlib style fallback
p = mod_path("scripts.create_tables")
txt = backup_and_read(p)
txt = re.sub(
    r'plt\.style\.use\("seaborn"\)',
    'try:\n    plt.style.use("seaborn-v0_8")\nexcept OSError:\n    plt.style.use("default")',
    txt,
    count=1
)
p.write_text(txt)

print("GenoVi patching complete")
PY
```


#### Correction of GenBank flatfile generated by Geneious
When exporting the annotated assembly from Geneious in GenBank flatfile format: 
- Select: Convert annotations to strict GenBank format
- Check yes: Generate new _translation_ qualifiers for truncated CDS annotations
- Uncheck: Allow names longer than 16 characters

The GenBank flatfile generated by Geneious likely needs to be reconstituted. The following code does this:

```bash
for GENOME in chromosome plasmid; do
    IN_GB="./input/Limnothrix_sp_HT2024_${GENOME}.gb"
    OUT_GB="./input/Limnothrix_sp_HT2024_${GENOME}_corrected.gb"

    case "$GENOME" in
        chromosome) SHORT_ID="Limnothrix_chr" ;;
        plasmid)    SHORT_ID="Limnothrix_pls" ;;
    esac

    awk -v sid="$SHORT_ID" '
    /^LOCUS/ {
        n=$3+0
        topo=($6 != "" ? $6 : "linear")
        div=($7 != "" ? $7 : "BCT")
        dt=($8 != "" ? toupper($8) : strftime("%d-%b-%Y"))
        printf("LOCUS       %-16s %11d bp    DNA     %-8s %-3s %s\n", sid, n, topo, div, dt)
        l=sid
        next
    }
    /^ACCESSION/ {print "ACCESSION   " l; next}
    /^VERSION/ {print "VERSION     " l ".1"; next}
    /^FEATURES[ \t]+Location\/Qualifiers/ {
        print
        print "     source          1.." n
        print "                     /organism=\"" l "\""
        print "                     /mol_type=\"genomic DNA\""
        next
    }
    {
        print
        last=$0
        gsub(/\r/, "", last)
        sub(/[[:space:]]+$/, "", last)
    }
    END {if (last != "//") print "//"}
    ' "$IN_GB" > "$OUT_GB"
done
```


#### (Optional) Adding translations to GenBank flatfile
For COGs to be identified and visualized by GenoVi, the input GenBank flatfile must contain translation qualifiers for every CDS. Oftentimes, these are not present in the GenBank flatfile. The custom script `add_translations_to_gb.py` adds them. Hence, the script must be executed before genoVi is run.

```bash
for GENOME in chromosome plasmid; do
    IN_GB="./input/Limnothrix_sp_HT2024_${GENOME}_corrected.gb"
    WO_GB="./input/Limnothrix_sp_HT2024_${GENOME}_corrected_woTranslations.gb"

    if [ -f "$IN_GB" ] && [ ! -f "$WO_GB" ]; then
        mv "$IN_GB" "$WO_GB"
    fi

    [ -f "$WO_GB" ] || { echo "SKIP: $WO_GB not found"; continue; }

  python3 add_translations_to_gb.py \
            -i "$WO_GB" \
            -o "$IN_GB" \
      --table 11 --overwrite
done
```

#### Execution of GenoVi
```bash
conda activate genovi

GENOME="chromosome"
#GENOME="plasmid"

# Clean temporary files from previous run
rm -rf conf etc fonts circos.conf circos.svg circos.png genovi-temp circos.log circos.debug.log 2>/dev/null

INPUT_GB="./input/Limnothrix_sp_HT2024_${GENOME}_corrected.gb"
[ -f "$INPUT_GB" ] || INPUT_GB="./input/Limnothrix_sp_HT2024_${GENOME}_corrected_woTranslations.gb"
[ -f "$INPUT_GB" ] || { echo "ERROR: no corrected input found for $GENOME"; exit 1; }

# Minimal GenBank preflight
sed -i 's/\r$//' "$INPUT_GB"
if ! awk 'NF{last=$0} END{exit(last=="//"?0:1)}' "$INPUT_GB"; then
    echo "WARN: missing final // in $INPUT_GB; appending terminator"
    printf "\n//\n" >> "$INPUT_GB"
fi
grep -q '^LOCUS' "$INPUT_GB" || { echo "ERROR: missing LOCUS in $INPUT_GB"; exit 1; }
grep -q '^ORIGIN' "$INPUT_GB" || { echo "ERROR: missing ORIGIN in $INPUT_GB"; exit 1; }

# Run GenoVi
genovi -i "$INPUT_GB" -cs strong -s complete -t Limnothrix_sp_HT2024_${GENOME} -te --size -k -v verbose

# Copy key outputs (if produced)
[ -f ./genovi/genovi.svg ] && cp ./genovi/genovi.svg "Limnothrix_sp_HT2024_${GENOME}__output_from_GenoVi.svg"
[ -f ./genovi/genovi_COG_Histogram.png ] && cp ./genovi/genovi_COG_Histogram.png "Limnothrix_sp_HT2024_${GENOME}__COG_Histogram.png"
[ -f ./genovi/genovi.svg ] || echo "GenoVi failed before final SVG creation; inspect circos.log and circos.debug.log"

# Generate COG barchart from GenoVi CSV
BARPLOT_OUT="Limnothrix_sp_HT2024_${GENOME}__COG_barplot.svg"
CSV_IN=""
for f in ./genovi/genovi_COG_Classification.csv ./genovi/genovi_COGclassification.csv; do
    if [ -f "$f" ]; then
        CSV_IN="$f"
        break
    fi
done
if [ -z "$CSV_IN" ]; then
    echo "WARN: missing GenoVi COG classification CSV; skipped COG barplot generation."
elif ! python3 plot_cog_barchart.py -i "$CSV_IN"; then
    echo "WARN: plot_cog_barchart.py failed; no COG barplot SVG written."
elif [ ! -f "${CSV_IN%.csv}_COG_barplot.svg" ]; then
    echo "WARN: plot_cog_barchart.py finished but no SVG was generated."
else
    cp "${CSV_IN%.csv}_COG_barplot.svg" "$BARPLOT_OUT"
    echo "INFO: wrote $BARPLOT_OUT"
    if [ -f "Limnothrix_sp_HT2024_${GENOME}__COG_Histogram.png" ]; then
        rm -f "Limnothrix_sp_HT2024_${GENOME}__COG_Histogram.png"
        echo "INFO: removed Limnothrix_sp_HT2024_${GENOME}__COG_Histogram.png (replaced by barchart output)."
    fi
fi

# Archive run artifacts (skip paths that do not exist)
target_dir="genoVi_${GENOME}"
mkdir -p "$target_dir"
to_archive=(circos.conf circos.debug.log circos.log circos.svg conf etc fonts genovi genovi-temp)
for p in "${to_archive[@]}"; do
    if [ -e "$p" ]; then
        mv "$p" "$target_dir"/
    fi
done
tar czf "${target_dir}.tar.gz" "$target_dir"
echo "INFO: archived run to ${target_dir}.tar.gz"
```
